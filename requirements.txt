# pyuvc must come from git: the PyPI wheel (pupil-labs-uvc 1.0.0b7) is built
# against an older libuvc and fails to open any camera.  See setup/install.sh
# and docs/USB-BANDWIDTH.md.
pupil-labs-uvc @ git+https://github.com/pupil-labs/pyuvc

# Only used by the diagnostic paths that talk to the STWIN board directly.
pyusb

# The GNSS receiver is a USB CDC-ACM serial port; gpslog talks to it through
# this.  It was missing here while being imported by three modules, so a fresh
# install ran the cameras and failed at the GPS logger.
pyserial

# Shrinks preview frames before they go out over Wi-Fi (pupilrec/preview.py).
# Without it the preview still works, at full size -- so it is worth having, but
# nothing breaks on a host that cannot install it.
Pillow
